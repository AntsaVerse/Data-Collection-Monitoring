  [new Date().toISOString().slice(0, 10)],   // ← tableau avec UNE SEULE date : aujourd'hui
  {
    label: html`<div style="font-weight:600; font-size:16px;">Date de la collecte</div>`,

    value: new Date().toISOString().slice(0, 10) 
  }
)

filteredData = collecte_jour.filter(d => d.today === dateDropdown)

  "Nom du Projet": d.project_name,
  "Organisation": d.organization,
  "Collecte Communauté Hôte": d.collecte_com_hote,
  "Collecte PDI": d.collecte_pdi,
  "Collecte Réfugiés": d.collecte_refugie,
  "Collecte Totale": d.collecte_total
}));

 html`
<div style="font-size:16px;">
  ${Inputs.table(tableData, {sort: "Collecte Totale", reverse: true})}
</div>
